async function processInventory() { // ============================================================ // READ VALUES FROM THE FORM // ============================================================ const partNumber = document.getElementById("partNumber").value.trim(); const location = document.getElementById("location").value.trim(); const toLocation = document.getElementById("toLocation").value.trim(); const quantity = Number(document.getElementById("quantity").value); // ============================================================ // VALIDATE USER INPUT // Stop immediately if something important is missing. // ============================================================ if (!selectedAction) { document.getElementById("status").innerText = "Please choose ADD, REMOVE, or MOVE."; return; } // end selectedAction check if (!partNumber) { document.getElementById("status").innerText = "Please enter or scan a part number."; return; } // end part number check if (!location) { document.getElementById("status").innerText = selectedAction === "move" ? "Please enter the FROM location." : "Please enter or scan a location."; return; } // end location check if ( selectedAction === "move" && !toLocation ) { document.getElementById("status").innerText = "Please enter the TO location."; return; } // end MOVE destination check if ( selectedAction === "move" && location === toLocation ) { document.getElementById("status").innerText = "FROM and TO locations cannot be the same."; return; } // end same-location check if (!quantity || quantity <= 0) { document.getElementById("status").innerText = "Please enter a valid quantity."; return; } // end quantity check // ============================================================ // PREVENT DOUBLE SUBMISSION // Disable the button while the request is being processed. // ============================================================ const processButton = document.getElementById("processButton"); processButton.disabled = true; processButton.innerText = "PROCESSING..."; // Remove any old success/error styling before starting. document.getElementById("status").classList.remove( "status-success", "status-error" ); // Show the operator what the system is doing. document.getElementById("status").innerText = selectedAction === "add" ? "Adding inventory..." : selectedAction === "remove" ? "Removing inventory..." : "Moving inventory..."; // ============================================================ // BUILD THE REQUEST URL // This sends the inventory request to Google Apps Script. // ============================================================ let url = WEB_APP_URL + "?action=" + encodeURIComponent(selectedAction) + "&partNumber=" + encodeURIComponent(partNumber) + "&location=" + encodeURIComponent(location) + "&quantity=" + encodeURIComponent(quantity); // MOVE needs one extra value: the destination location. if (selectedAction === "move") { url += "&toLocation=" + encodeURIComponent(toLocation); } // end MOVE URL addition // ============================================================ // SEND REQUEST TO GOOGLE APPS SCRIPT // ============================================================ try { const response = await fetch(url); const data = await response.json(); // ========================================================== // HANDLE SUCCESS // ========================================================== if (data.success) { document.getElementById("status") .classList.remove("status-error"); document.getElementById("status") .classList.add("status-success"); // -------------------------------------------------------- // ADD SUCCESS // -------------------------------------------------------- if (selectedAction === "add") { document.getElementById("status").innerText = "ADD COMPLETE\n" + "Part: " + partNumber + "\n" + "Location: " + location + "\n" + "Quantity added: " + quantity + "\n" + "Available now: " + data.quantity; document.getElementById("available").innerText = "Available: " + data.quantity; // -------------------------------------------------------- // REMOVE SUCCESS // -------------------------------------------------------- } else if (selectedAction === "remove") { document.getElementById("status").innerText = "REMOVE COMPLETE\n" + "Part: " + partNumber + "\n" + "Location: " + location + "\n" + "Quantity removed: " + quantity + "\n" + "Available now: " + data.quantity; document.getElementById("available").innerText = "Available: " + data.quantity; // -------------------------------------------------------- // MOVE SUCCESS // -------------------------------------------------------- } else if (selectedAction === "move") { document.getElementById("status").innerText = "MOVE COMPLETE\n" + "Part: " + partNumber + "\n" + "From: " + location + "\n" + "To: " + toLocation + "\n" + "Quantity moved: " + quantity + "\n" + "Available at destination: " + data.toQuantity; document.getElementById("available").innerText = "Available at " + toLocation + ": " + data.toQuantity; } // end selectedAction success handling // Clear form fields after a successful transaction. resetFormAfterSuccess(); // ========================================================== // HANDLE API ERROR // The request worked, but the backend rejected the action. // Example: trying to remove more inventory than exists. // ========================================================== } else { document.getElementById("status") .classList.remove("status-success"); document.getElementById("status") .classList.add("status-error"); document.getElementById("status").innerText = "ERROR\n" + data.message; } // end data.success check // ============================================================ // HANDLE CONNECTION OR JAVASCRIPT ERRORS // ============================================================ } catch (error) { document.getElementById("status") .classList.remove("status-success"); document.getElementById("status") .classList.add("status-error"); document.getElementById("status").innerText = "CONNECTION ERROR\n" + error.message; // ============================================================ // ALWAYS RUN THIS // Whether success or failure, re-enable the button. // ============================================================ } finally { processButton.disabled = false; processButton.innerText = "PROCESS INVENTORY"; } // end try / catch / finally } // end processInventory()
